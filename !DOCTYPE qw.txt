<script>
/* ============ helpers ============ */
const $ = sel => document.querySelector(sel);
const $$ = sel => document.querySelectorAll(sel);
const screens = {};
$$('.screen').forEach(s=> screens[s.id]=s);
function showScreen(id){
  Object.values(screens).forEach(s=>s.classList.remove('active'));
  screens[id].classList.add('active');
  $('#backBtn').classList.toggle('show', id!=='screen-start');
}

/* floating hearts background */
const floaters = $('#floaters');
const heartChars = ['❤','✦','♡','✧'];
for(let i=0;i<16;i++){
  const s = document.createElement('span');
  s.textContent = heartChars[Math.floor(Math.random()*heartChars.length)];
  s.style.left = Math.random()*100+'%';
  s.style.fontSize = (12+Math.random()*16)+'px';
  s.style.animationDuration = (10+Math.random()*14)+'s';
  s.style.animationDelay = (Math.random()*14)+'s';
  floaters.appendChild(s);
}

/* music */
const music = $('#bgMusic');
const musicBtn = $('#musicToggle');
let musicStarted = false;
musicBtn.addEventListener('click', ()=>{
  if(music.paused){ music.play().catch(()=>{}); musicBtn.classList.remove('muted'); musicBtn.textContent='♪'; }
  else { music.pause(); musicBtn.classList.add('muted'); musicBtn.textContent='✕'; }
});
function tryStartMusic(){
  if(musicStarted) return;
  musicStarted = true;
  music.volume = 0.55;
  music.play().catch(()=>{ musicBtn.classList.add('muted'); musicBtn.textContent='♪'; });
}

const chatScripts = {
  Hian: [
    {who:'them', text:'Oi, amor... vi que você entrou aqui. Que saudade de você. ❤'},
    {who:'them', text:'Eu sei que moramos longe e ainda nem nos vimos pessoalmente, mas a gente conversa todo santo dia.'},
    {who:'them', text:'Fiz esse joguinho pensando em você. Quer jogar comigo?'},
  ],
  Richard: [
    {who:'them', text:'Oi, meu amor. Que bom te ver por aqui de novo. ❤'},
    {who:'them', text:'Sei que a distância pesa às vezes, mas cada conversa nossa faz valer a pena esperar.'},
    {who:'them', text:'Preparei um cantinho nosso aqui. Bora jogar?'},
  ]
};
function startChat(char){
  $('#chatTitle').textContent = char === 'Hian' ? 'Richard está te esperando' : 'Hian está te esperando';
  const box = $('#chatMessages');
  box.innerHTML = '';
  $('#chatReplies').innerHTML = '';
  const script = chatScripts[char];
  let i = 0;
  function next(){
    if(i < script.length){
      addBubble(script[i].who, script[i].text);
      i++;
      setTimeout(next, 900);
    } else {
      showReplies();
    }
  }
  setTimeout(next, 500);
}
function addBubble(who, text){
  const box = $('#chatMessages');
  const b = document.createElement('div');
  b.className = 'bubble ' + who;
  b.textContent = text;
  box.appendChild(b);
  box.scrollTop = box.scrollHeight;
}
function showReplies(){
  const rep = $('#chatReplies');
  rep.innerHTML = '';
  const opts = ['Bora! ❤', 'Estou com saudade também...'];
  opts.forEach(o=>{
    const btn = document.createElement('button');
    btn.className = 'btn small';
    btn.textContent = o;
    btn.addEventListener('click', ()=>{
      addBubble('me', o);
      rep.innerHTML = '';
      setTimeout(()=>{
        addBubble('them', 'Então vem comigo. 💛');
        setTimeout(()=>{ updateMenuGreeting(); showScreen('screen-menu'); }, 1100);
      }, 700);
    });
    rep.appendChild(btn);
  });
}
function updateMenuGreeting(){
  $('#menuGreeting').textContent = chosenChar ? `${chosenChar}, o que vamos jogar hoje?` : 'o que vamos jogar hoje?';
}

/* menu navigation */

    if(g==='quiz') startQuiz();
    if(g==='seq') startSeq();
    if(g==='puzzle') startPuzzle();
    if(g==='memory') startMemory();
  });
});
$('#endMenu').addEventListener('click', ()=>{ updateMenuGreeting(); showScreen('screen-menu'); });

/* ============ MAZE GAME (10 levels, generated labyrinths) ============ */
let mazeLevel = 1, mazeData = null, mazeSize = 5, playerPos = {x:0,y:0}, goalPos = {x:0,y:0};

function startMaze(){
  mazeLevel = 1;
  showScreen('screen-maze');
  buildMazeLevel();
}
function buildMazeLevel(){
  mazeSize = 7 + Math.floor((mazeLevel-1) * 1.4); // cresce de 7 até ~19, bem mais difícil
  if(mazeSize % 2 === 0) mazeSize++;
  $('#mazeLevel').textContent = `fase ${mazeLevel} / 10`;
  $('#mazeStatus').textContent = 'chegue até o coração ❤';
  mazeData = generateMaze(mazeSize, mazeSize);
  playerPos = {x:0, y:0};
  goalPos = {x:mazeSize-1, y:mazeSize-1};
  renderMaze();
}
// generate maze via recursive backtracker; grid cells: 1=wall,0=path. Dimensions are (2n+1)x(2n+1)
function generateMaze(cols, rows){
  const w = cols, h = rows; // logical cell counts
  const gridW = w*2+1, gridH = h*2+1;
  const grid = Array.from({length:gridH}, ()=> Array(gridW).fill(1));
  const visited = Array.from({length:h}, ()=> Array(w).fill(false));
  function carve(cx, cy){
    visited[cy][cx] = true;
    grid[cy*2+1][cx*2+1] = 0;
    const dirs = shuffleArray([[0,-1],[0,1],[-1,0],[1,0]]);
    for(const [dx,dy] of dirs){
      const nx = cx+dx, ny = cy+dy;
      if(nx>=0 && nx<w && ny>=0 && ny<h && !visited[ny][nx]){
        grid[cy*2+1+dy][cx*2+1+dx] = 0;
        carve(nx, ny);
      }
    }
  }
  carve(0,0);
  return grid; // size gridH x gridW, indices [row][col], 0=path,1=wall
}
function renderMaze(){
  const wrap = $('#mazeWrap');
  const gridH = mazeData.length, gridW = mazeData[0].length;
  const maxBox = Math.min(window.innerWidth - 56, window.innerHeight * 0.42, 340);
  const cellPx = Math.max(6, Math.min(maxBox / gridW, maxBox / gridH));
  wrap.innerHTML = '';
  const grid = document.createElement('div');
  grid.className = 'maze-grid';
  grid.style.gridTemplateColumns = `repeat(${gridW}, ${cellPx}px)`;
  grid.style.gridTemplateRows = `repeat(${gridH}, ${cellPx}px)`;
  for(let r=0;r<gridH;r++){
    for(let c=0;c<gridW;c++){
      const cell = document.createElement('div');
      cell.style.width = cellPx+'px';
      cell.style.height = cellPx+'px';
      const isWall = mazeData[r][c]===1;
      const px = (c-1)/2, py = (r-1)/2;
      const isPlayer = !isWall && px===playerPos.x && py===playerPos.y;
      const isGoal = !isWall && px===goalPos.x && py===goalPos.y;
      cell.className = 'mcell ' + (isWall?'wall':'path') + (isPlayer?' player':'') + (isGoal?' goal':'');
      if(isGoal && !isPlayer) cell.textContent = '❤';
      grid.appendChild(cell);
    }
  }
  wrap.appendChild(grid);
}
function canMove(px, py, dx, dy){
  // check wall between cells in grid coordinates
  const gx = px*2+1+dx, gy = py*2+1+dy;
  if(gy<0 || gy>=mazeData.length || gx<0 || gx>=mazeData[0].length) return false;
  return mazeData[gy][gx] === 0;
}
$$('.maze-btn').forEach(b=>{
  b.addEventListener('click', ()=>{
    const d = b.dataset.d;
    let dx=0, dy=0;
    if(d==='up') dy=-1; if(d==='down') dy=1; if(d==='left') dx=-1; if(d==='right') dx=1;
    if(canMove(playerPos.x, playerPos.y, dx, dy)){
      playerPos.x += dx; playerPos.y += dy;
      renderMaze();
      if(playerPos.x===goalPos.x && playerPos.y===goalPos.y){
        if(mazeLevel >= 10){
          setTimeout(()=> showEnd('Você chegou até o coração!', 'Completou as 10 fases do labirinto da saudade ❤', '❤'), 300);
        } else {
          $('#mazeStatus').textContent = 'muito bem! próxima fase...';
          mazeLevel++;
          setTimeout(buildMazeLevel, 800);
        }
      }
    }
  });
});

/* ============ PUZZLE (3x3 sliding heart puzzle) ============ */
const puzzleSymbols = ['1','2','3','4','5','6','7','8',''];
let puzzleState = [];
function startPuzzle(){
  showScreen('screen-puzzle');
  $('#puzzleStatus').textContent = 'arraste as peças até organizar de 1 a 8';
  do{
    puzzleState = shuffleArray([...puzzleSymbols]);
  } while(!isSolvable(puzzleState));
  renderPuzzle();
}
function shuffleArray(arr){
  const a = [...arr];
  for(let i=a.length-1;i>0;i--){
    const j = Math.floor(Math.random()*(i+1));
    [a[i],a[j]]=[a[j],a[i]];
  }
  return a;
}
function isSolvable(arr){
  const nums = arr.map(s=> s===''? 9 : puzzleSymbols.indexOf(s)+1);
  let inv=0;
  for(let i=0;i<nums.length;i++) for(let j=i+1;j<nums.length;j++)
    if(nums[i]!==9 && nums[j]!==9 && nums[i]>nums[j]) inv++;
  return inv % 2 === 0;
}
function renderPuzzle(){
  const grid = $('#puzzleGrid');
  grid.innerHTML = '';
  puzzleState.forEach((sym, idx)=>{
    const t = document.createElement('div');
    t.className = 'ptile' + (sym===''?' empty':'');
    t.textContent = sym;
    t.addEventListener('click', ()=> tryMove(idx));
    grid.appendChild(t);
  });
}
function tryMove(idx){
  const empty = puzzleState.indexOf('');
  const r1=Math.floor(idx/3), c1=idx%3, r2=Math.floor(empty/3), c2=empty%3;
  if(Math.abs(r1-r2)+Math.abs(c1-c2)===1){
    [puzzleState[idx], puzzleState[empty]] = [puzzleState[empty], puzzleState[idx]];
    renderPuzzle();
    if(puzzleState.join('') === puzzleSymbols.join('')){
      setTimeout(()=> showEnd('Coração completo!', 'Você montou o nosso quebra-cabeça 🧩', '🧩'), 300);
    }
  }
}

/* ============ MEMORY GAME (turn-based: você vs bot, com rodadas) ============ */
const memoryIconPool = ['❤','✨','💌','🌙','💍','🎀','🌹','💫','☾','🕊','🌷','🦋'];
let memCards = [], memMatched = 0, memLock = false, memFlipped = [];
let memRound = 1, memPairsTotal = 6;
let memTurn = 'p1'; // 'p1' = jogador, 'p2' = bot
let memScores = {p1:0, p2:0};
let botMemory = {}; // icon -> [idx,...] seen by bot

function otherChar(){
  return chosenChar === 'Hian' ? 'Richard' : (chosenChar === 'Richard' ? 'Hian' : 'Bot');
}

function startMemory(){
  memRound = 1;
  memScores = {p1:0, p2:0};
  $('#p1Name').textContent = chosenChar || 'Você';
  $('#p2Name').textContent = otherChar();
  showScreen('screen-memory');
  setupMemoryRound();
}
function setupMemoryRound(){
  memPairsTotal = Math.min(5 + memRound, memoryIconPool.length); // cresce a cada rodada
  $('#memRound').textContent = `rodada ${memRound}`;
  memMatched = 0; memFlipped = []; memLock = false; botMemory = {};
  memTurn = 'p1';
  updateTurnUI();
  const icons = shuffleArray([...memoryIconPool]).slice(0, memPairsTotal);
  memCards = shuffleArray([...icons, ...icons]);
  const grid = $('#memoryGrid');
  grid.innerHTML = '';
  memCards.forEach((icon, idx)=>{
    const c = document.createElement('div');
    c.className = 'mcard';
    c.dataset.icon = icon;
    c.dataset.idx = idx;
    c.textContent = '?';
    c.addEventListener('click', ()=> flipCard(c, true));
    grid.appendChild(c);
  });
}
function updateTurnUI(){
  $('#p1Score').textContent = memScores.p1;
  $('#p2Score').textContent = memScores.p2;
  $('#turnP1').classList.toggle('active', memTurn==='p1');
  $('#turnP2').classList.toggle('active', memTurn==='p2');
  $('#memTurnLabel').textContent = memTurn==='p1' ? 'sua vez' : `vez de ${otherChar()}...`;
}
function rememberCard(c){
  const icon = c.dataset.icon, idx = parseInt(c.dataset.idx);
  if(!botMemory[icon]) botMemory[icon] = [];
  if(!botMemory[icon].includes(idx)) botMemory[icon].push(idx);
}
function flipCard(c, byPlayer){
  if(memLock || c.classList.contains('flipped') || c.classList.contains('matched')) return;
  if(byPlayer && memTurn !== 'p1') return;
  c.classList.add('flipped');
  c.textContent = c.dataset.icon;
  rememberCard(c);
  memFlipped.push(c);
  if(memFlipped.length===2){
    memLock = true;
    const [a,b] = memFlipped;
    if(a.dataset.icon === b.dataset.icon){
      a.classList.add('matched'); b.classList.add('matched');
      memMatched++;
      memScores[memTurn]++;
      memFlipped = []; memLock = false;
      updateTurnUI();
      if(memMatched === memPairsTotal){
        finishMemoryRound();
      } else {
        // acertou: joga de novo (mesma vez)
        if(memTurn === 'p2') setTimeout(botTurn, 700);
      }
    } else {
      setTimeout(()=>{
        a.classList.remove('flipped'); a.textContent='?';
        b.classList.remove('flipped'); b.textContent='?';
        memFlipped = []; memLock = false;
        // errou: passa a vez
        memTurn = memTurn === 'p1' ? 'p2' : 'p1';
        updateTurnUI();
        if(memTurn === 'p2') setTimeout(botTurn, 700);
      }, 750);
    }
  }
}
function botTurn(){
  if(memTurn !== 'p2') return;
  const cards = [...$('#memoryGrid').children].filter(c=> !c.classList.contains('matched') && !c.classList.contains('flipped'));
  if(cards.length === 0) return;
  // tenta usar memória: ver se conhece um par já visto
  let pick1 = null, pick2 = null;
  for(const icon in botMemory){
    const idxs = botMemory[icon].filter(i=>{
      const el = $(`.mcard[data-idx="${i}"]`);
      return el && !el.classList.contains('matched');
    });
    if(idxs.length >= 2 && Math.random() < 0.75){
      pick1 = $(`.mcard[data-idx="${idxs[0]}"]`);
      pick2 = $(`.mcard[data-idx="${idxs[1]}"]`);
      break;
    }
  }
  if(pick1 && pick2){
    flipCard(pick1, false);
    setTimeout(()=> flipCard(pick2, false), 550);
  } else {
    const choice1 = cards[Math.floor(Math.random()*cards.length)];
    flipCard(choice1, false);
    setTimeout(()=>{
      const remaining = [...$('#memoryGrid').children].filter(c=> !c.classList.contains('matched') && !c.classList.contains('flipped') && c!==choice1);
      // tenta achar o par conhecido da carta que acabou de virar
      const icon = choice1.dataset.icon;
      let matchEl = null;
      if(botMemory[icon]){
        const idxs = botMemory[icon].filter(i=> i!==parseInt(choice1.dataset.idx));
        for(const i of idxs){
          const el = $(`.mcard[data-idx="${i}"]`);
          if(el && !el.classList.contains('matched')){ matchEl = el; break; }
        }
      }
      const choice2 = matchEl || remaining[Math.floor(Math.random()*remaining.length)];
      flipCard(choice2, false);
    }, 600);
  }
}
function finishMemoryRound(){
  setTimeout(()=>{
    const total = memScores.p1 + memScores.p2;
    if(memRound >= 4){
      const vencedor = memScores.p1 > memScores.p2 ? (chosenChar||'Você') : (memScores.p2 > memScores.p1 ? otherChar() : 'empate');
      const sub = vencedor === 'empate'
        ? `Empate de ${memScores.p1} a ${memScores.p2} depois de ${memRound} rodadas 🃏`
        : `${vencedor} venceu por ${Math.max(memScores.p1,memScores.p2)} a ${Math.min(memScores.p1,memScores.p2)} 🃏`;
      showEnd('Fim de jogo!', sub, '🃏');
    } else {
      memRound++;
      setupMemoryRound();
    }
  }, 500);
}
